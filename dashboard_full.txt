def safe_read_csv(path):
    """Safely load a CSV file."""

    if path is None:
        return None

    path = Path(path)

    if not path.exists():
        return None

    try:
        return pd.read_csv(path)
    except Exception as exc:
        st.warning(
            f"Could not load {path.name}: {exc}"
        )
        return None

                    matching.iloc[0][
